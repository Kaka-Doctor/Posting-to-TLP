     1→     1→#!/usr/bin/env node
     2→     2→/**
     3→     3→ * ─────────────────────────────────────────────────────────────────────────────
     4→     4→ *  Social Media Auto-Posting Agent — standalone GitHub Actions runner
     5→     5→ * ─────────────────────────────────────────────────────────────────────────────
     6→     6→ *  Runs on GitHub's servers (`.github/workflows/social-agent.yml`), triggered
     7→     7→ *  by GitHub's cron. No web server, no database, no external image host, no
     8→     8→ *  npm installs — plain Node (>= 18) reading the repo itself.
     9→     9→ *
    10→    10→ *  RELIABILITY MODEL: GitHub silently drops most scheduled runs on young
    11→    11→ *  accounts (observed live: 1 run per 5-11 hours, regardless of cron density).
    12→    12→ *  The agent therefore does NOT rely on firing at slot time. Every landed run
    13→    13→ *  PRE-SCHEDULES the next `scheduleDays` days of slots with Meta's own
    14→    14→ *  scheduled publishing (IG containers with publish_time, FB photos with
    15→    15→ *  scheduled_publish_time) — Meta then publishes them exactly on time, from
    16→    16→ *  Meta's servers. One landed run every ~2 days is enough. Landed runs also
    17→    17→ *  reconcile state once scheduled posts go live, and late direct posting
    18→    18→ *  remains as the catch-up fallback.
    19→    19→ *
    20→    20→ *  Everything configurable lives in data/config.json (the switch file):
    21→    21→ *    pageId / igUserId  → which page/account to post to
    22→    22→ *    slots / timezone   → when to post (cron here just checks often)
    23→    23→ *    pageAccessToken    → permanent page token (preferred)
    24→    24→ *    metaUserToken      → user token, used to re-derive a page token if the
    25→    25→ *                         permanent one ever stops working (rotation path)
    26→    26→ *
    27→    27→ *  Image pipeline (fully Meta-internal, no public hosting needed):
    28→    28→ *    Instagram requires a public image_url. We upload the photo to the FB page
    29→    29→ *    as an UNPUBLISHED photo (binary upload, invisible), take its public CDN
    30→    30→ *    URL, feed that to the IG container, publish on IG, then delete the hidden
    31→    31→ *    photo. Facebook itself is posted via direct binary upload.
    32→    32→ *
    33→    33→ *  State lives in data/state.json and is committed back to the repo by the
    34→    34→ *  workflow (which slots are done per day + rotation counters + event log).
    35→    35→ *
    36→    36→ *  Flags:  --dry-run   pick content, print the plan, publish nothing
    37→    37→ *          --force     post the next scheduled slot right now (manual runs)
    38→    38→ *  Exit:   0 = ok / nothing to do, 1 = failure (red run in the Actions tab)
    39→    39→ */

    58→    58→async function graphGet(pathname, params) {
    59→    59→  const url = new URL(`${GRAPH}/${pathname}`)
    60→    60→  for (const [k, v] of Object.entries(params)) url.searchParams.set(k, String(v))
    61→    61→  const res = await fetch(url, { method: 'GET' })
    62→    62→  const json = await res.json().catch(() => ({}))
    63→    63→  if (!res.ok) throw new Error(json?.error?.message ?? `HTTP ${res.status}`)
    64→    64→  return json
    65→    65→}

    67→    67→async function graphPost(pathname, params) {
    68→    68→  const body = new URLSearchParams()
    69→    69→  for (const [k, v] of Object.entries(params)) body.set(k, String(v))
    70→    70→  const res = await fetch(`${GRAPH}/${pathname}`, { method: 'POST', body })
    71→    71→  const json = await res.json().catch(() => ({}))
    72→    72→  if (!res.ok || json.error) throw new Error(json?.error?.message ?? `HTTP ${res.status}`)
    73→    73→  return json
    74→    74→}

    86→    86→async function fbUploadPhoto(pageId, pageToken, filePath, { caption = '', published = true } = {}) {
    87→    87→  const buf = await fs.readFile(filePath)
    88→    88→  const form = new FormData()
    89→    89→  if (caption) form.append('caption', caption)
    90→    90→  form.append('published', published ? 'true' : 'false')
    91→    91→  form.append('access_token', pageToken)
    92→    92→  form.append('source', new Blob([buf], { type: 'image/jpeg' }), path.basename(filePath))
    93→    93→  const res = await fetch(`${GRAPH}/${pageId}/photos`, { method: 'POST', body: form })
    94→    94→  const json = await res.json().catch(() => ({}))
    95→    95→  if (!res.ok || json.error) throw new Error(json?.error?.message ?? `HTTP ${res.status}`)
    96→    96→  return json
    97→    97→}

    99→    99→// ── Connect (page-token first, user-token exchange as fallback) ─────────────
   100→   100→async function connect(cfg) {
   101→   101→  // 1) permanent page token from config
   102→   102→  if (cfg.pageAccessToken && cfg.pageId) {
   103→   103→    try {
   104→   104→      const me = await graphGet(cfg.pageId, {
   105→   105→        fields: 'name,instagram_business_account{id,username}',
   106→   106→        access_token: cfg.pageAccessToken,
   107→   107→      })
   108→   108→      const ig = me.instagram_business_account ?? {}
   109→   109→      log(`[connect] page token (permanent): valid → ${me.name} | IG: @${ig.username ?? 'none'}`)
   110→   110→      return {
   111→   111→        via: 'page-token',
   112→   112→        page: {
   113→   113→          pageId: cfg.pageId,
   114→   114→          pageName: String(me.name ?? ''),
   115→   115→          pageToken: cfg.pageAccessToken,
   116→   116→          igUserId: String(ig.id ?? cfg.igUserId ?? ''),
   117→   117→          igUsername: String(ig.username ?? ''),
   118→   118→        },
   119→   119→      }
   120→   120→    } catch (e) {
   121→   121→      log(`[connect] permanent page token failed (${e.message}) — trying user-token exchange`)
   122→   122→    }
   123→   123→  }
   124→   124→  // 2) rotation path: exchange user token → pages → page token
   125→   125→  if (!cfg.metaUserToken) throw new Error('No working token: pageAccessToken invalid and metaUserToken missing — regenerate your Meta token and update data/config.json')
   126→   126→  let token = cfg.metaUserToken
   127→   127→  if (cfg.appId && cfg.appSecret) {
   128→   128→    try {
   129→   129→      const ex = await graphGet('oauth/access_token', {
   130→   130→        grant_type: 'fb_exchange_token',
   131→   131→        client_id: cfg.appId,
   132→   132→        client_secret: cfg.appSecret,
   133→   133→        fb_exchange_token: token,
   134→   134→      })
   135→   135→      if (ex.access_token) token = String(ex.access_token)
   136→   136→    } catch { /* keep raw token */ }
   137→   137→  }
   138→   138→  const pages = await graphGet('me/accounts', {
   139→   139→    fields: 'id,name,access_token,instagram_business_account{id,username}',
   140→   140→    access_token: token,
   141→   141→    limit: '50',
   142→   142→  })
   143→   143→  const list = (pages.data ?? []).map((p) => ({
   144→   144→    id: String(p.id),
   145→   145→    name: String(p.name),
   146→   146→    accessToken: String(p.access_token),
   147→   147→    igUserId: p.instagram_business_account ? String(p.instagram_business_account.id) : '',
   148→   148→    igUsername: p.instagram_business_account ? String(p.instagram_business_account.username) : '',
   149→   149→  }))
   150→   150→  const page =
   151→   151→    list.find((p) => cfg.pageId && p.id === cfg.pageId) ??
   152→   152→    list.find((p) => cfg.igUsername && p.igUsername === cfg.igUsername) ??
   153→   153→    list.find((p) => p.igUserId) ??
   154→   154→    list[0]
   155→   155→  if (!page) throw new Error('Token exchange OK but no pages visible to this token')
   156→   156→  log(`[connect] user-token path → ${page.name} (${page.id}) | IG: @${page.igUsername}`)
   157→   157→  return { via: 'exchange', page: { ...page, pageToken: page.accessToken } }
   158→   158→}

   161→   161→function localNow(timeZone) {
   162→   162→  const parts = new Intl.DateTimeFormat('en-CA', {
   163→   163→    timeZone,
   164→   164→    hourCycle: 'h23',
   165→   165→    year: 'numeric',
   166→   166→    month: '2-digit',
   167→   167→    day: '2-digit',
   168→   168→    hour: '2-digit',
   169→   169→    minute: '2-digit',
   170→   170→  }).formatToParts(new Date())
   171→   171→  const get = (t) => parts.find((p) => p.type === t)?.value ?? '0'
   172→   172→  const hour = Number(get('hour')) % 24
   173→   173→  const minute = Number(get('minute'))
   174→   174→  return {
   175→   175→    date: `${get('year')}-${get('month')}-${get('day')}`,
   176→   176→    minutesOfDay: hour * 60 + minute,
   177→   177→  }
   178→   178→}

   180→   180→function parseSlots(str) {
   181→   181→  return String(str || '06:00,12:00,18:00')
   182→   182→    .split(',')
   183→   183→    .map((s) => s.trim().match(/^(\d{1,2}):(\d{2})$/))
   184→   184→    .filter(Boolean)
   185→   185→    .map((m) => ({
   186→   186→      minutes: Number(m[1]) * 60 + Number(m[2]),
   187→   187→      label: `${String(Number(m[1])).padStart(2, '0')}:${m[2]}`,
   188→   188→    }))
   189→   189→    .sort((a, b) => a.minutes - b.minutes)
   190→   190→}

   193→   193→async function loadLibrary() {
   194→   194→  const photos = (await fs.readdir(PHOTOS_DIR))
   195→   195→    .filter((f) => /\.jpe?g$/i.test(f))

   198→   198→  if (nonJpg.length > 0) {
   199→   199→    log(`[library] note: ${nonJpg.length} non-JPEG file(s) present (${nonJpg.join(', ')}) — the workflow's normalize step converts them to JPEG; IG can only post JPEG`)
   200→   200→  }
   201→   201→  const captions = Array.from(
   202→   202→    new Set(
   203→   203→      (await fs.readFile(CAPTIONS_FILE, 'utf-8'))
   204→   204→        .split(/\n\s*---+\s*\n/)
   205→   205→        .map((t) => t.trim())
   206→   206→        .filter((t) => t.length >= 2 && t.length <= 2200)
   207→   207→    )
   208→   208→  )
   209→   209→  return { photos, captions }
   210→   210→}

   230→   230→// ── Duplicate guard (mirrors src/lib/meta.ts minutesSinceLastPost) ──────────
   231→   231→async function minutesSinceLastPost(page) {
   232→   232→  const cutoff = Date.now() - 7 * 86400000
   233→   233→  const future = Date.now() + 60000 // ignore future timestamps (scheduled posts)
   234→   234→  try {
   235→   235→    if (page.igUserId) {
   236→   236→      const json = await graphGet(`${page.igUserId}/media`, { fields: 'timestamp', limit: '3', access_token: page.pageToken })
   237→   237→      const ts = (json.data ?? []).map((m) => (m.timestamp ? new Date(m.timestamp).getTime() : 0)).filter(Boolean).filter((t) => t < future)
   238→   238→      const latest = Math.max(...ts, 0)
   239→   239→      if (latest > cutoff) return Math.max(0, Math.round((Date.now() - latest) / 60000))
   240→   240→      return null
   241→   241→    }
   242→   242→  } catch { /* fall through */ }
   243→   243→  try {
   244→   244→    const json = await graphGet(`${page.pageId}/posts`, { fields: 'created_time', limit: '3', access_token: page.pageToken })
   245→   245→    const ts = (json.data ?? []).map((p) => (p.created_time ? new Date(p.created_time).getTime() : 0)).filter(Boolean).filter((t) => t < future)
   246→   246→    const latest = Math.max(...ts, 0)
   247→   247→    if (latest > cutoff) return Math.max(0, Math.round((Date.now() - latest) / 60000))
   248→   248→  } catch { /* unknown — do not block */ }
   249→   249→  return null
   250→   250→}

   252→   252→// ── Publish (IG via Meta CDN, FB via binary upload) ─────────────────────────
   253→   253→async function publishInstagram(page, photoPath, caption) {
   254→   254→  const steps = []
   255→   255→  let helperPhotoId = ''
   256→   256→  try {
   257→   257→    // 1) hidden FB photo → public CDN URL
   258→   258→    const up = await fbUploadPhoto(page.pageId, page.pageToken, photoPath, { published: false })
   259→   259→    helperPhotoId = String(up.id ?? '')
   260→   260→    steps.push(`helper photo ${helperPhotoId} uploaded (unpublished)`)
   261→   261→    const info = await graphGet(helperPhotoId, { fields: 'images', access_token: page.pageToken })
   262→   262→    const images = (info.images ?? []).slice().sort((a, b) => (b.width ?? 0) - (a.width ?? 0))
   263→   263→    const cdnUrl = String(images[0]?.source ?? '')
   264→   264→    if (!cdnUrl) throw new Error('no CDN URL on the helper photo')
   265→   265→    steps.push(`CDN url ok (${images[0].width}px)`)
   266→   266→
   267→   267→    // 2) IG container → poll → publish (mirrors src/lib/meta.ts)
   268→   268→    const container = await graphPost(`${page.igUserId}/media`, {
   269→   269→      image_url: cdnUrl,
   270→   270→      caption: caption.slice(0, 2200),
   271→   271→      access_token: page.pageToken,
   272→   272→    })
   273→   273→    const creationId = String(container.id ?? '')
   274→   274→    if (!creationId) throw new Error('no container id returned')
   275→   275→    steps.push(`container ${creationId}`)
   276→   276→    let ready = false
   277→   277→    for (let i = 0; i < 10; i++) {
   278→   278→      const status = await graphGet(creationId, { fields: 'status_code', access_token: page.pageToken })
   279→   279→      const code = String(status.status_code ?? '')
   280→   280→      if (code === 'FINISHED') { ready = true; break }
   281→   281→      if (code === 'ERROR') throw new Error('Instagram could not process the image')
   282→   282→      await new Promise((r) => setTimeout(r, 2000))
   283→   283→    }
   284→   284→    if (!ready) throw new Error('image processing timed out (try a smaller JPEG)')
   285→   285→    steps.push('container FINISHED')
   286→   286→
   287→   287→    const published = await graphPost(`${page.igUserId}/media_publish`, { creation_id: creationId, access_token: page.pageToken })
   288→   288→    const postId = String(published.id ?? '')
   289→   289→    let permalink = ''
   290→   290→    try {
   291→   291→      const info2 = await graphGet(postId, { fields: 'permalink', access_token: page.pageToken })
   292→   292→      permalink = String(info2.permalink ?? '')
   293→   293→    } catch { /* best-effort */ }
   294→   294→    steps.push(`published ${postId}`)
   295→   295→    return { ok: true, postId, permalink, error: '', steps }
   296→   296→  } catch (e) {
   297→   297→    return { ok: false, postId: '', permalink: '', error: e.message, steps }
   298→   298→  } finally {
   299→   299→    if (helperPhotoId) {
   300→   300→      try { await graphDelete(helperPhotoId, { access_token: page.pageToken }) } catch { /* best-effort cleanup */ }
   301→   301→    }
   302→   302→  }
   303→   303→}

   384→   384→  } catch (e) {
   385→   385→    return { ok: false, postId: '', error: e.message }
   386→   386→  }
   387→   387→}

   389→   389→/** Confirm Meta-scheduled slots that have gone live → record permalinks. */
   390→   390→async function reconcileScheduled(page, state) {
   391→   391→  let changed = false
   392→   392→  for (const [dk, day] of Object.entries(state.days ?? {})) {
   393→   393→    for (const [label, rec] of Object.entries(day ?? {})) {
   394→   394→      if (rec?.status !== 'scheduled' || !rec.publishTime) continue
   395→   395→      const due = new Date(rec.publishTime).getTime()
   396→   396→      if (Date.now() < due + 5 * 60000) continue // not due yet
   397→   397→      let igOk = false, fbOk = false, igPerma = '', fbPerma = ''
   398→   398→      if (rec.ig?.id) {
   399→   399→        try {
   400→   400→          const m = await graphGet(String(rec.ig.id), { fields: 'permalink,timestamp', access_token: page.pageToken })
   401→   401→          const live = m.permalink && m.timestamp && new Date(m.timestamp).getTime() <= Date.now() + 60000
   402→   402→          if (live) { igOk = true; igPerma = String(m.permalink) }
   403→   403→        } catch { /* not confirmable yet */ }
   404→   404→      }
   405→   405→      if (rec.fb?.id) {
   406→   406→        // photo id while unpublished; pageId_photoId once it becomes a feed post
   407→   407→        for (const cid of [String(rec.fb.id), `${page.pageId}_${String(rec.fb.id)}`]) {
   408→   408→          try {
   409→   409→            const p = await graphGet(cid, { fields: 'permalink_url,created_time', access_token: page.pageToken })
   410→   410→            const live = p.permalink_url && p.created_time && new Date(p.created_time).getTime() <= Date.now() + 60000
   411→   411→            if (live) { fbOk = true; fbPerma = String(p.permalink_url); rec.fb.id = cid; break }
   412→   412→          } catch { /* not confirmable via this id yet */ }
   413→   413→        }
   414→   414→      }
   415→   415→      if (igOk || fbOk) {
   416→   416→        rec.status = igOk && fbOk ? 'published' : 'partial'
   417→   417→        rec.at = rec.publishTime
   418→   418→        if (rec.ig) rec.ig = { id: rec.ig.id, permalink: igPerma }
   419→   419→        if (rec.fb) rec.fb = { id: rec.fb.id, permalink: fbPerma }
   420→   420→        recordEvent(state, { level: igOk && fbOk ? 'success' : 'error', date: dk, slot: label, message: `${label} published by Meta on schedule — IG ${igOk ? `✓ ${igPerma}` : '✗ unconfirmed'} | FB ${fbOk ? `✓ ${fbPerma}` : '✗ unconfirmed'}` })
   421→   421→        changed = true
   422→   422→      }
   423→   423→    }
   424→   424→  }
   425→   425→  return changed
   426→   426→}

   428→   428→/** Pause switch: delete pending scheduled posts from Meta so "autoPost": false really stops everything. */
   429→   429→async function cancelPendingScheduled(page, state) {
   430→   430→  let cancelled = 0
   431→   431→  for (const [dk, day] of Object.entries(state.days ?? {})) {
   432→   432→    for (const [label, rec] of Object.entries(day ?? {})) {
   433→   433→      if (rec?.status !== 'scheduled' || !rec.publishTime) continue
   434→   434→      if (new Date(rec.publishTime).getTime() < Date.now()) continue
   435→   435→      const removed = []
   436→   436→      if (rec.ig?.id) { try { await graphDelete(String(rec.ig.id), { access_token: page.pageToken }); removed.push('IG') } catch { /* already gone */ } }
   437→   437→      if (rec.fb?.id) { try { await graphDelete(String(rec.fb.id), { access_token: page.pageToken }); removed.push('FB') } catch { /* already gone */ } }
   438→   438→      if (removed.length > 0) {
   439→   439→        rec.status = 'cancelled'
   440→   440→        rec.at = new Date().toISOString()
   441→   441→        recordEvent(state, { level: 'info', date: dk, slot: label, message: `${label} cancelled (autoPost=false) — removed from Meta: ${removed.join(' + ')}` })
   442→   442→        cancelled++
   443→   443→      }
   444→   444→    }
   445→   445→  }
   446→   446→  return cancelled
   447→   447→}

   449→   449→/** Wall-clock (dateKey + minutesOfDay) in `timeZone` → UTC epoch ms. 2-pass Intl convergence. */
   450→   450→function zonedToUtc(dateKey, minutesOfDay, timeZone) {
   451→   451→  const [y, mo, d] = dateKey.split('-').map(Number)
   452→   452→  const wallMs = Date.UTC(y, mo - 1, d, Math.floor(minutesOfDay / 60), minutesOfDay % 60)
   453→   453→  const fmt = new Intl.DateTimeFormat('en-CA', { timeZone, hourCycle: 'h23', year: 'numeric', month: '2-digit', day: '2-digit', hour: '2-digit', minute: '2-digit' })
   454→   454→  let ts = wallMs
   455→   455→  for (let i = 0; i < 3; i++) {
   456→   456→    const parts = fmt.formatToParts(new Date(ts))
   457→   457→    const get = (t) => Number(parts.find((p) => p.type === t)?.value ?? 0)
   458→   458→    const asUTC = Date.UTC(get('year'), get('month') - 1, get('day'), get('hour') % 24, get('minute'))
   459→   459→    if (asUTC === wallMs) break
   460→   460→    ts += wallMs - asUTC
   461→   461→  }
   462→   462→  return ts
   463→   463→}

   465→   465→/** Upcoming slots across `scheduleDays` days that have no state entry yet. */
   466→   466→function buildHorizon(state, slots, now, tz, scheduleDays) {
   467→   467→  const plan = []
   468→   468→  const minUnix = Date.now() + 20 * 60000 // Meta needs ≥ 10 min lead; use 20 for safety
   469→   469→  for (let d = 0; d < scheduleDays; d++) {
   470→   470→    const dk = d === 0 ? now.date : addDaysKey(now.date, d)
   471→   471→    for (const s of slots) {
   472→   472→      if (state.days[dk]?.[s.label]) continue
   473→   473→      const slotUnix = zonedToUtc(dk, s.minutes, tz)
   474→   474→      if (slotUnix < minUnix) continue
   475→   475→      plan.push({ dk, slot: s, slotUnix })
   476→   476→    }
   477→   477→  }
   478→   478→  return plan
   479→   479→}

   702→   702→        ig: igS.ok ? { id: igS.postId } : { error: igS.error },
   703→   703→        fb: fbS.ok ? { id: fbS.postId } : { error: fbS.error },
   704→   704→      }
   705→   705→      state.days[h.dk] = day
   706→   706→      state.rotation.photoIndex = (state.rotation.photoIndex + 1) % photos.length
   707→   707→      state.rotation.captionIndex = (state.rotation.captionIndex + 1) % captions.length
   708→   708→      recordEvent(state, { level: 'info', date: h.dk, slot: h.slot.label, message: `${h.slot.label} pre-scheduled via Meta (publishes ${hISO}) — "${hPhoto}" | IG ${igS.ok ? '✓' : `✗ ${igS.error}`} | FB ${fbS.ok ? '✓' : `✗ ${fbS.error}`}` })

   710→   710→      scheduledCount++
   711→   711→    } else {
   712→   712→      recordEvent(state, { level: 'error', date: h.dk, slot: h.slot.label, message: `${h.slot.label} scheduling FAILED — IG: ${igS.error} | FB: ${fbS.error} (next run retries)` })
   713→   713→      log(`  ✗ scheduling failed: IG ${igS.error} | FB ${fbS.error} — next run retries`)
   714→   714→    }
   715→   715→  }
   716→   716→  if (scheduledCount > 0) {
   717→   717→    await writeState(state)
   718→   718→    log(`[state] saved — ${scheduledCount} upcoming slot(s) pre-scheduled with Meta`)
   719→   719→  }
   720→   720→}
